ITEM15: ROUTE
    first_cruising_speed_and_level +
    Route:
        Flights along designated ATS routes:
            1: ats_route_name or (DCT + point_name + ats_route_name)
            2: points_at_which:
                change_of_speed_level_or_flight_rules or change of ats_routes
            3: ats_route_name(even if the same as the previous one) or DCT
        Flights outside designated ATS routes:
            1: point_name or refrence_to_point
            2:  DCT between successive points unless both points are defined by geographical coordinates or by bearing and distance.

    cruising speed:
        (?P<c_speed>(K(0|1)[\d]{3})|(N0[\d]{3})|(M(0[\d]{2}|10\d)))
    cruising level:
        (?P<c_level>(F[\d]{2}[0,5])|(S[\d]{3}0)|(A[\d]{2}[0,5])|(M[\d]{4}))
    point name:
        (?P<ats_point>[A-Z]{2,5})
    ats route:
        (?P<ats_route>[A-Z]{1,3}[\d]{1,3})
        (?P<sid_star>(?P=ats_point)[\d][A-Z])
    degrees:
        (?P<degrees>(([0-8]\d)|90)(N|S)((0[\d]{2})|(1[0-7]\d)|180)(E|W))
    degrees_and_minutes
        (?P<degrees_and_minutes>(([0-8]\d)|90)[0-5]\d(N|S)((0[\d]{2})|(1[0-7]\d)|180)[0-5]\d(E|W))
